import numpy as np

# Build a periodic tone. amps[m] is the amplitude of the (m+1)-th harmonic
def additive(t, f0, amps, fs):

    x = np.zeros_like(t)
    for m in range(len(amps)):
        k = m + 1
        if k*f0 < fs/2:
            x = x + amps[m] * np.sin(2 * np.pi * k * f0 * t) # Complete: add the k-th harmonic, see Eq. (1) of the manual
    return x

# only the fundamental is nonzero
pure = np.zeros(20)
pure[0] = 1

# odd harmonics with amplitude 1/k, even harmonics 0
odd_1k = np.zeros(20)
for i in range(len(odd_1k)):
    k = i + 1
    if k % 2 != 0:
        odd_1k[i] = float(1/k)

# all harmonics with amplitude 1/k 
all_1k = np.zeros(20)
for i in range(len(all_1k)):
    k = i + 1
    all_1k[i] = float(1/k)

# all harmonics with amplitude 1/k**2
all_1k2 = np.zeros(20)
for i in range(len(all_1k2)):
    k = i + 1
    all_1k2[i] = float((1/k) ** 2)

tone_pure = additive(t, f0, pure, fs) 
tone_pure = tone_pure / np.max(np.abs(tone_pure)) #normalize to [-1, 1]

tone_odd_1k = additive(t, f0, odd_1k, fs)
tone_odd_1k = tone_odd_1k / np.max(np.abs(tone_odd_1k)) #normalize to [-1, 1]

tone_all_1k = additive(t, f0, all_1k, fs)
tone_all_1k = tone_all_1k / np.max(np.abs(tone_all_1k)) #normalize to [-1, 1]

tone_all_1k2 = additive(t, f0, all_1k2, fs)
tone_all_1k2 = tone_all_1k2 / np.max(np.abs(tone_all_1k2)) #normalize to [-1, 1]

#Write tones to files to playback and compare

\subsection*{Brightness and Hollowness}
\begin{description}
  \item[\texttt{pure}:] Only the fundamental is nonzero, so there is no higher-frequency content. It sounds the least bright of the four, with a very smooth but dull overall sound.
  \item[\texttt{odd\_1k}:] Only the odd harmonics are nonzero, with amplitudes decaying as $1/k$. It sounds very bright, and also the hollowest of the four.

  \item[\texttt{all\_1k}:] All harmonics are present, with amplitudes decaying as $1/k$. It sounds bright, but with a fuller sound compared to odd\_1k.
  \item[\texttt{all\_1k2}:] All harmonics are present, but the amplitudes decay as $1/k^2$, so the higher harmonics die out much sooner. It sounds mellower than the $1/k$ tones.
\end{description}

The brightest tones are the ones whose harmonic amplitudes decay as $1/k$ (\texttt{odd\_1k} and \texttt{all\_1k}), because they contain more high-frequency content. The pure tone has no high-frequency content, and in \texttt{all\_1k2} the $1/k^2$ decay makes the higher harmonics fall off much faster. Hollowness is more of a result of the harmonic pattern: the hollowest tone, \texttt{odd\_1k}, has only odd harmonics. Generally, tones built from only odd harmonics sound hollower than other tones (such as square waves).

\subsection*{Tone Shape}
\begin{description}
  \item[\texttt{pure}:] The \texttt{pure} tone looks like an ordinary sine wave, since it is built from the fundamental alone with no additional frequency content.
  \item[\texttt{odd\_1k}:] The \texttt{odd\_1k} tone looks like a square wave. This matches the Fourier series of a square wave, whose coefficients are zero for even harmonics and proportional to $1/k$ for odd harmonics.
  \item[\texttt{all\_1k}:] The \texttt{all\_1k} tone looks like a sawtooth wave, consistent with the Fourier series of a sawtooth, whose coefficients are proportional to $1/k$ for all harmonics $k \geq 1$.
  \item[\texttt{all\_1k2}:] The \texttt{all\_1k2} waveform appears to lie between a sawtooth and a pure sine wave, just as the magnitude plot of its coefficients lies between those of the sawtooth tone (\texttt{all\_1k}) and the pure tone (\texttt{pure}).
\end{description}

The pattern of coefficients determines the waveform's shape in two ways. First, which harmonics are present sets the overall profile: odd harmonics alone produce a square wave, while all harmonics with the same $1/k$ decay produce a sawtooth. Second, how fast the coefficient decays determines how sharp the features of the waveform are, with faster decaying coefficient leading to a smoother, more sine-like waveform.

import numpy as np
import matplotlib.pyplot as plt
from scipy.io import wavfile

fs = 8192

# Build a periodic tone. amps[m] is the amplitude of the (m+1)-th harmonic
def additive(t, f0, amps, fs):
    x = np.zeros_like(t)
    for m in range(len(amps)):
        k = m + 1
        if k*f0 < fs/2:
            x = x + amps[m] * np.sin(2 * np.pi * k * f0 * t) # Complete: add the k-th harmonic, see Eq. (1) of the manual
    return x

def envelope(t, t_a=0.01, tau_d=0.15):
    # returns t/t_a during the rise (t < t_a), exp decay afterwards
    return np.where(t < t_a, t/t_a, np.exp(-(t-t_a)/tau_d))

freqs = {"C4":261.63, "D4":293.66, "E4":329.63, "F4":349.23, "G4":392.00, "A4":440.00}

song = [("C4",0.5),("C4",0.5),("G4",0.5),("G4",0.5),("A4",0.5),("A4",0.5),("G4",1.0),("F4",0.5),("F4",0.5),("E4",0.5),("E4",0.5),("D4",0.5),("D4",0.5),("C4",1.0)]
my_amps = [1, 1.3, 1, 0.9, 0, 0.65, 0.6]

clip = []
for name, duration in song:
    tn = np.arange(0, duration, 1.0/fs)
    note = additive(tn, freqs[name], my_amps, fs)

    clip.append(note * envelope(tn))

clip = np.concatenate(clip)
clip = clip/np.max(np.abs(clip))
clip_i16 = np.int16(clip * 32767) # converts to 16 bit int for wav format
wavfile.write("clip.wav", round(fs), clip_i16) # writes the wav file to disk
\end{minted}

import numpy as np
import matplotlib.pyplot as plt
from scipy.io import wavfile

A = 1.0
f0 = 10.0
w0 = 2*np.pi*f0
T0 = 1.0/f0

def xtilde(t, N):
    x = np.zeros_like(t)
    for n in range(1, N+1):
        k = 2*n- 1
        x = x + 1/k * np.sin(k * w0 * t)
    return x * (4 * A)/np.pi


def square_wave(t, T):
    sq = np.zeros_like(t)
    for i in range(len(t)):
        rem = t[i] % T
        if rem < T/2:
            sq[i] = A
        else: 
            sq[i] = -A

    return sq

t = np.arange(-0.25*T0, 1.25*T0, T0/10000)

N = [1, 3, 7, 49, 199]

fig1, ax1 = plt.subplots(figsize=(10, 5))

for n in N:
    ax1.plot(t, xtilde(t, n), label=f'xtilde with N = {n}')

peak_N49 = np.max(xtilde(t, 49))
peak_N199 = np.max(xtilde(t, 199))

ax1.plot(t, square_wave(t, T0), label='Ideal square wave')

ax1.set_xlim(-0.25 * T0, 1.25 * T0)
ax1.set_title('Truncated Square Wave Plot, with Approximations')
ax1.set_xlabel('t (s)')
ax1.set_ylabel('amplitude')
ax1.legend()

fig1.tight_layout()
plt.show()
\end{minted}

\subsubsection*{Overshoot Measurement and Behavior}
\addcontentsline{toc}{subsubsection}{Overshoot Measurement and Behavior}
For a $N=49$, the peak value just after the jump is $1.1790$, exceeding $A=1$ by \SI{17.9}{\percent}, while for $N=199$, the peak value just after the jump is $1.1778$, exceeding $A=1$ by \SI{17.78}{\percent}. For both cases, the exceeded percentage is approximately the same, even though over four times as many harmonics were used for $N=199$. This shows the difficulty of representing a discontinuous signal with a finite number of harmonics, as it will not be possible to perfectly approximate the discontinuities.
